33def ecrire_si_change(chemin, texte):
34 """
35 @brief Ecrit un fichier seulement si son contenu change.
36 @param chemin : fichier a ecrire.
37 @param texte : nouveau contenu.
38 @retval True si le fichier a ete reecrit.
39 @par Exemple d'utilisation
40 ecrire_si_change("Core/Inc/ldo_config_genere.h", h)
41 """
42 if os.path.isfile(chemin):
43 with open(chemin, "r", encoding="utf-8") as f:
44 if f.read() == texte:
45 return False
46 with open(chemin, "w", encoding="utf-8", newline="\n") as f:
47 f.write(texte)
48 return True

51def main():
52 """
53 @brief Verifie le CSV et genere les fichiers C.
54 @retval Code de sortie : 0 = OK, 1 = erreur (la compilation s'arrete).
55 @par Exemple d'utilisation
56 python configurationLDO/generer_config.py --verifier
57 """
58 if not os.path.isfile(CSV):
59 print("%s:1: error: fichier introuvable" % CSV)

62 ioc = iocs[0] if iocs else None
63
64 lignes, erreurs_format = cfg.lire_fichier(CSV)
65 erreurs, avertissements = cfg.verifier(lignes, ioc) if not erreurs_format else ([], [])
66 erreurs = erreurs_format + erreurs
67
68 # Format "fichier:ligne: error:" : CubeIDE l'affiche dans l'onglet Problems.
69 for a in avertissements:
70 print("%s:1: warning: %s" % (CSV, a))
71 for e in erreurs:
72 print("%s:1: error: %s" % (CSV, e))
73 if erreurs:
74 print("generer_config : %d erreur(s), configuration refusee, rien n'est genere." % len(erreurs))
75 return 1
76
77 if "--verifier" in sys.argv:
78 print("generer_config : %d LDO, configuration valide, empreinte 0x%08X." % (len(lignes), cfg.empreinte(lignes)))
79 return 0
80
81 h, inc = cfg.generer_c(lignes)
82 modifs = [os.path.basename(p) for p, t in ((SORTIE_H, h), (SORTIE_INC, inc)) if ecrire_si_change(p, t)]
83 print("generer_config : %d LDO, empreinte 0x%08X, %s." % (
84 len(lignes), cfg.empreinte(lignes), ("mis a jour : " + ", ".join(modifs)) if modifs else "deja a jour"))
85 return 0
