254// Valeur d'initialisation de chaque sortie DAC au demarrage ET valeur restauree
255// par la boucle periodique de StartLwipTask() apres tout code operateur qui ne
256// passe pas par VAL_LDO[]/ldoConfig[].ldo->VAL (ex: juste apres un balayage a
257// distance BALAYER_DAC, dont la cible n'est pas forcement une LDO configuree) :
258// 0V, jamais une tension intermediaire, par securite (aucune sortie ne doit
259// "sortir" une tension non voulue par l'operateur simplement parce qu'elle n'a
260// jamais ete explicitement configuree).

267// ---------------------------------------------------------------------------
268// Protection surintensite (OC) des LDO equipees du circuit shunt R97 +
269// OP184 + LM211 (aujourd'hui 1PSp et 2PSp). Revision du 13/10/2026.
270//
271// CARTE_LIM_CORRIGEE : 1 = carte corrigee (R96 = R98 = 26,1 kOhm 0,1 %,
272// C178 = C356 = 10 nF C0G, R103 = 10 kOhm, RH = 680 kOhm),
273// 0 = carte actuelle (R96 = R98 = 100 kOhm, 100 nF).
274// C'est le SEUL define a changer pour passer d'une carte a l'autre.
275// ---------------------------------------------------------------------------
276#define CARTE_LIM_CORRIGEE 1
277
278#if CARTE_LIM_CORRIGEE
279#define OC_K_V_PAR_A 2.61f // G x R97 = 26,1 x 0,1 Ohm (V/A)
280#define OC_BLANKING_MS 10U // soft-start + 5 tau (tau = 261 us)
281#define OC_CODE_SEUIL_MAX 4006U // 1,90 A : pire cas 1,951 A < 2,0 A
282#define OC_CODE_SEUIL_DEFAUT 1056U // 0,50 A au demarrage (hysteresis incluse)
283#else
284#define OC_K_V_PAR_A 10.0f // G x R97 = 100 x 0,1 Ohm (V/A)
285#define OC_BLANKING_MS 60U // soft-start + 5 tau (tau = 10 ms)
286#define OC_CODE_SEUIL_MAX 4095U // 0,50 A, pleine echelle du DAC
287#define OC_CODE_SEUIL_DEFAUT 4095U // 0,50 A au demarrage (500 mA = 4096, ramene a 4095)
288#endif
289
290// Code minimal du seuil : 0,5 V = limite basse d'entree du LM211
291// (0,5 x 4096 / 5 = 409,6 -> 410), identique pour les deux cartes.
292#define OC_CODE_SEUIL_MIN 410U
293
294// Niveau de la broche OC qui signifie "surintensite". Schema PS2+ : IN+ du
295// LM211 = VMES (sortie OP184 via R103), IN- = DAC_OUT, donc OC = 1 en
296// surintensite. A passer a GPIO_PIN_RESET uniquement si l'essai T5 montre
297// la polarite inverse.
298#define OC_NIVEAU_DEFAUT GPIO_PIN_SET
299
300// ---------------------------------------------------------------------------
301// Source du seuil de limitation de courant (revision du 21/10/2026).
302// 0 = seuil fixe par le DAC de courant (MCP47FEB28, voie "DAC courant" de la
303// ligne de ldoConfig[] : 0xC0 voie 5 pour 2PSp). Cas de la carte actuelle.
304// 1 = seuil fixe par un potentiometre (R104 / RV2 montes, DAC_OUT deconnecte
305// de IN- du LM211) : la voie du DAC de courant est mise en haute impedance
306// (Power-Down 11), SET_CURRENT_LDO_ est refuse et l'EN n'attend plus de
307// seuil DAC valide.
308// Valeur par defaut de la colonne "seuil" de ldoConfig[] (SEUIL_SOURCE) : une
309// LDO peut aussi etre forcee individuellement a SEUIL_DAC ou SEUIL_POT.
310// ---------------------------------------------------------------------------
311#define LIMITATION_PAR_POTENTIOMETRE 0
312
313// ---------------------------------------------------------------------------
314// Memoire des codes DAC (tension + seuil de courant) dans l'EEPROM du
315// MCP47FEB28 (registres non volatils 10h-17h), pour les LDO declarees MEM_EEPROM
316// dans ldoConfig[]. Au demarrage le DAC recharge lui-meme ces valeurs et le
317// firmware les relit ; chaque nouvelle valeur (SET_VAL_LDO_, SET_CURRENT_LDO_)
318// est ecrite apres DAC_EEPROM_DELAI_MS sans changement (une rampe n'use pas
319// l'EEPROM), et immediatement apres un declenchement OC.
320// DAC_EEPROM_TWC_MS : duree d'ecriture EEPROM, tWC = 16 ms max (DS20006368A).
321// ---------------------------------------------------------------------------
322#define DAC_EEPROM_TWC_MS 20U
323#define DAC_EEPROM_DELAI_MS 2000U
324#define DAC_EEPROM_ESSAIS_MAX 3U
